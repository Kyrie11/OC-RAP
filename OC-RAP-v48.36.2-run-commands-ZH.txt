- RC=20：pipeline 有效但 gate 拒绝，可进行算法层分析；
- RC=30：工程失败，查看 PIPELINE_FAILED.json 和对应阶段日志。

# 仅当第三步发现 checkpoint 缺失或 repair contract 拒绝 checkpoint 字节时使用。
FRESH_OUTPUTDIR=runs/ocrap_v48_36_2_ocaf_stage_transfer_hotfix_48362
mkdir -p "$FRESH_OUTPUTDIR"

# 可以只复用通过 contract 的 teacher index/summary；不要复制 checkpoint 或状态文件。
for f in \
  evidence_adapt_teacher_pcd_index.jsonl \
  evidence_adapt_teacher_pcd_index_summary.json \
  evidence_adapt_dev_teacher_pcd_index.jsonl \
  evidence_adapt_dev_teacher_pcd_index_summary.json; do
  [[ -f "$OUTPUTDIR/$f" ]] && cp -f "$OUTPUTDIR/$f" "$FRESH_OUTPUTDIR/$f"

# 更新 summary 中的 output provenance。
python - "$FRESH_OUTPUTDIR" <<'PY_REBASE'
import json, os, pathlib, sys, time
root=pathlib.Path(sys.argv[1])
for summary_name,index_name in (
    ('evidence_adapt_teacher_pcd_index_summary.json','evidence_adapt_teacher_pcd_index.jsonl'),
    ('evidence_adapt_dev_teacher_pcd_index_summary.json','evidence_adapt_dev_teacher_pcd_index.jsonl'),
):
    path=root/summary_name
    if not path.is_file():
        continue
    doc=json.loads(path.read_text())
    doc['output']=str(root/index_name)
    doc['copied_for_v48_36_2_fresh_rerun']=True
    doc['copied_unix']=time.time()
    tmp=path.with_name(f'.{path.name}.tmp.{os.getpid()}')
    tmp.write_text(json.dumps(doc,ensure_ascii=False,indent=2)+'\n')
    os.replace(tmp,path)
PY_REBASE

set +e

# 默认仍不需要上传大 checkpoint。
DIAG_DIR="$OUTPUTDIR/v48_36_2_minimal_diagnostics"
rm -rf "$DIAG_DIR"; mkdir -p "$DIAG_DIR"

for f in \
  ATTEMPT_STARTED.json \
  V48_36_2_STAGE_TRANSFER_REPAIR.json \
  V48_36_RESUME_CONTRACT.json \
  PIPELINE_FAILED.json \
  V48_36_COMPLETE.json \
  AUTHORITATIVE_RUN_STATUS.json \
  NEXT_COMMANDS_STATUS.json; do
  [[ -f "$OUTPUTDIR/$f" ]] && cp -a "$OUTPUTDIR/$f" "$DIAG_DIR/"
done

mkdir -p "$DIAG_DIR/candidates" "$DIAG_DIR/logs"
for variant in balanced precision; do
  mkdir -p "$DIAG_DIR/candidates/$variant"
  for f in \
    STAGE_TRANSFER_INTEGRITY.json \
    STAGE_TRANSFER_REPAIR_REJECTED.v48.36.2.json \
    THREE_STAGE_TRAINING_COMPLETE.json \

    [[ -f "$OUTPUTDIR/candidates/$variant/$f" ]] && \
      cp -a "$OUTPUTDIR/candidates/$variant/$f" "$DIAG_DIR/candidates/$variant/"
  done
done

for f in \
  resume_contract.log \
  training_contract_balanced.log \
  training_contract_precision.log \
  model_contract_balanced.log \
  model_contract_precision.log \
  certificate_controller.log \
