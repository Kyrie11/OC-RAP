七、严格解释 RC
----------------------------------------------------------------------
case "$RC" in
  0)
    echo "有效通过：检查 V48_35_COMPLETE.json，然后只执行 NEXT_COMMANDS.txt 中注册的 Safe/stress/消融。"
    test -f "$OUTPUTDIR/V48_35_COMPLETE.json"
    test -f "$OUTPUTDIR/NEXT_COMMANDS.txt"
    cat "$OUTPUTDIR/NEXT_COMMANDS.txt"
    ;;
  20)
    echo "有效算法拒绝：certificate/gate 已执行。不要读取 test；打包 certificate、shared rule、gate decomposition 后分析算法。"
    test -f "$OUTPUTDIR/V48_35_COMPLETE.json"
    test -f "$OUTPUTDIR/NEXT_COMMANDS_BLOCKED.json"
    ;;
  30)
    echo "仍为工程/协议失败：禁止做算法结论。优先检查 resume contract、PIPELINE_FAILED、training/model/certificate logs。" >&2
    test -f "$OUTPUTDIR/V48_35_RESUME_CONTRACT.json" && \
      python -m json.tool "$OUTPUTDIR/V48_35_RESUME_CONTRACT.json" || true
    test -f "$OUTPUTDIR/PIPELINE_FAILED.json" && \
      python -m json.tool "$OUTPUTDIR/PIPELINE_FAILED.json" || true
    ;;
  *)
    echo "非注册 RC=$RC；按工程错误处理。" >&2
    ;;
esac
exit "$RC"
