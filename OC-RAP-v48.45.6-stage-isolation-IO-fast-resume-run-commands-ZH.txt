#!/usr/bin/env bash
# OC-RAP v48.45.6 — SOWR stage-isolation + source-architecture contract + exact I/O fast path
#
# 默认是“工程恢复轮”：复用 hash/protocol 完全一致且 pipeline_valid 的 A (RC=0/20)，
# 自动删除并重跑 pipeline-invalid 的 B/C/D。这样不会把工程失败当算法结果，也避免重复浪费 A 的约 2 小时。
# 若要做论文最终归档的“同一 checkout 四臂全重跑”，把 FORCE_RERUN_VALID_ARMS=1。

set -Eeuo pipefail

# ============================================================
# 0. 只修改机器路径/GPU；不要修改算法超参、gate、top-k 或 harm budget
# ============================================================
export OCRAP_REPO=/home/senzeyu2/code/OC-RAP
export OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP
export BASE_OUT="$OCRAP_REPO/runs"
export SOURCE_RUN="$BASE_OUT/ocrap_v48_45_source_rebuild_s7"
export PROTOCOL_ROOT="$OCRAP_ROOT/calibration_v48_14_prism_4814"
export CAL_NEAR="$OCRAP_ROOT/calibration_near_contact"
export CAL_CONTACT="$OCRAP_ROOT/calibration_contact"
export CAL_SAFE="$OCRAP_ROOT/calibration_safe"
export GPU0=0
export GPU1=1

# 0 = 工程恢复轮：保留 hash-identical 的 authoritative RC=0/20 arm（推荐先这样）
# 1 = 论文最终归档轮：A/B/C/D 全部在 v48.45.6 checkout 下重跑
export FORCE_RERUN_VALID_ARMS="${FORCE_RERUN_VALID_ARMS:-0}"

# 性能参数：每个 arm 内仍是 balanced->GPU0, precision->GPU1；arm 间不并发。
# cache=true 只缓存“已解码后的模型输入/标签 CPU tensor”，不改变样本/损失/顺序。
export MAX_PARALLEL_ARMS=1
export ABLATION_CACHE_SAMPLES_IN_MEMORY=true
export ABLATION_NUM_WORKERS=3
export ABLATION_PREFETCH_FACTOR=3
export ABLATION_OMP_NUM_THREADS=1
export V48456_REUSE_AUTHORITATIVE_ARMS=1

cd "$OCRAP_REPO"
export PYTHONPATH="$OCRAP_REPO/src${PYTHONPATH:+:$PYTHONPATH}"
[[ "$GPU0" != "$GPU1" ]] || { echo 'ERROR GPU0/GPU1 must differ' >&2; exit 2; }

printf '\n========== 0/6 CODE REGRESSION + GPU PRECHECK ==========\n'
pytest -q \
  tests/test_v48_45_1_engineering_hotfix.py \
  tests/test_v48_45_sowr.py \
  tests/test_v48_45_5_protocol_bootstrap.py \
  tests/test_v48_45_6_stage_isolation_io.py
python -m compileall -q src tools
for f in scripts/*.sh; do bash -n "$f"; done
nvidia-smi -L

printf '\n========== 1/6 VERIFY SHARED CALIBRATION PROTOCOL ==========\n'
bash scripts/prepare_v48_45_protocol.sh 2>&1 | tee "$BASE_OUT/v48_45_6_protocol_prepare.log"
python - "$PROTOCOL_ROOT/V48_45_PROTOCOL_SEAL.json" <<'PY'
import json,sys
p=json.load(open(sys.argv[1]))
assert p.get('valid') is True and p.get('test_roots_read') is False,p
print('PROTOCOL SEAL PASS')
PY

printf '\n========== 2/6 VERIFY IMMUTABLE SHARED SOURCE ==========\n'
python tools/check_v48_36_source_checkpoint_contract.py \
  --source-run "$SOURCE_RUN" --output "$SOURCE_RUN/SOURCE_CHECKPOINT_CONTRACT_V48_45_6.json"
python tools/check_v48_45_rebuilt_source_quality.py \
  --source-run "$SOURCE_RUN" --output "$SOURCE_RUN/SOURCE_QUALITY_CONTRACT_V48_45_6.json"
python - "$SOURCE_RUN/SOURCE_CHECKPOINT_CONTRACT_V48_45_6.json" "$SOURCE_RUN/SOURCE_QUALITY_CONTRACT_V48_45_6.json" <<'PY'
import json,sys
h=json.load(open(sys.argv[1]));q=json.load(open(sys.argv[2]))
assert h.get('valid') is True,h
assert q.get('valid') is True,q
print('SOURCE HASH + QUALITY PASS')
for v in ('balanced','precision'): print(v,h['checks'][v]['sha256'])
PY

printf '\n========== 3/6 SAFE RESUME PLAN ==========\n'
resume_args=()
[[ "$FORCE_RERUN_VALID_ARMS" == 1 ]] && resume_args+=(--force-rerun-valid)
python tools/prepare_v48_45_6_resume.py \
  --base-out "$BASE_OUT" \
  --protocol-seal "$PROTOCOL_ROOT/V48_45_PROTOCOL_SEAL.json" \
  --source-run "$SOURCE_RUN" \
  --apply "${resume_args[@]}" \
  --output "$BASE_OUT/V48_45_6_RESUME_PLAN.json" \
  | tee "$BASE_OUT/v48_45_6_resume_plan.log"
# 预期当前上传结果对应的原机器状态：A=preserve_authoritative，B/C/D=remove_for_clean_retry。

printf '\n========== 4/6 RUN SOWR 2x2 (TWO GPUs, ONE ARM AT A TIME) ==========\n'
export SOURCE_RUN BASE_OUT PROTOCOL_ROOT CAL_NEAR CAL_CONTACT CAL_SAFE GPU0 GPU1
set +e
bash scripts/run_v48_45_sowr_2x2_parallel.sh \
  2>&1 | tee "$BASE_OUT/v48_45_6_sowr_2x2_driver.log"
launcher_rc=${PIPESTATUS[0]}
set -e
[[ -s "$BASE_OUT/ocrap_v48_45_sowr_parallel_status.json" ]] && cat "$BASE_OUT/ocrap_v48_45_sowr_parallel_status.json"
if [[ "$launcher_rc" != 0 ]]; then
  echo "ENGINEERING FAILURE remains: launcher_rc=$launcher_rc" >&2
  for r in ocrap_v48_45_sowr_ablation_A ocrap_v48_45_sowr_ablation_B ocrap_v48_45_sowr_ablation_C ocrap_v48_45_sowr_main; do
    echo "---- $r" >&2
    [[ -s "$BASE_OUT/$r/PIPELINE_FAILED.json" ]] && cat "$BASE_OUT/$r/PIPELINE_FAILED.json" >&2
    [[ -s "$BASE_OUT/$r/AUTHORITATIVE_RUN_STATUS.json" ]] && cat "$BASE_OUT/$r/AUTHORITATIVE_RUN_STATUS.json" >&2
  done
  exit 30
fi

printf '\n========== 5/6 REQUIRE FOUR AUTHORITATIVE RC=0/20 RESULTS ==========\n'
python - "$BASE_OUT/ocrap_v48_45_sowr_parallel_status.json" <<'PY'
import json,sys
s=json.load(open(sys.argv[1]))
assert s.get('engineering_failed') is False,s
for arm,row in s['arms'].items():
    assert row.get('launcher_exit_code') in (0,20),(arm,row)
print('ALL A/B/C/D ARE PIPELINE-VALID RC 0/20 RESULTS')
PY

printf '\n========== 6/6 ATTRIBUTION CONTRACT + 2x2 DIAGNOSTIC ==========\n'
python tools/compare_v48_45_sowr_2x2.py \
  --a "$BASE_OUT/ocrap_v48_45_sowr_ablation_A" \
  --b "$BASE_OUT/ocrap_v48_45_sowr_ablation_B" \
  --c "$BASE_OUT/ocrap_v48_45_sowr_ablation_C" \
  --d "$BASE_OUT/ocrap_v48_45_sowr_main" \
  --positive-gain 0.015 \
  --output "$BASE_OUT/ocrap_v48_45_sowr_2x2_comparison.json" \
  | tee "$BASE_OUT/v48_45_6_sowr_2x2_comparison.log"
python - "$BASE_OUT/ocrap_v48_45_sowr_2x2_comparison.json" "$FORCE_RERUN_VALID_ARMS" <<'PY'
import json,sys
x=json.load(open(sys.argv[1])); force=int(sys.argv[2])
contract=x.get('attribution_contract') or {}
assert contract.get('valid') is True,contract
print('ATTRIBUTION INPUT CONTRACT PASS: shared protocol/source/gate/dataset hashes identical')
if force == 1:
    print('PAPER-ATTRIBUTION EXECUTION CONTRACT PASS: all four arms were forced to rerun under v48.45.6')
else:
    print('RECOVERY-ROUND WARNING: a prior authoritative arm may have been reused. Use this round to verify engineering/directionality; run FORCE_RERUN_VALID_ARMS=1 before final paper-quality 2x2 attribution.')
PY

cat <<'EOF'

Interpretation:
  RC=30/other -> engineering/protocol failure; DO NOT analyze algorithmically.
  RC=20       -> pipeline_valid Natural-gate miss; valid negative algorithm result.
  RC=0        -> pipeline_valid gate pass.

Speed notes:
  - Keep MAX_PARALLEL_ARMS=1: each arm already occupies GPU0 and GPU1 simultaneously.
  - cache_samples_in_memory removes repeated compressed-NPZ decoding across epochs.
  - NUM_WORKERS=3/prefetch=3 is execution-only; if host RAM is constrained, first set
    ABLATION_CACHE_SAMPLES_IN_MEMORY=false rather than changing batch/loss/epochs.
  - Do not tune thresholds/top-k/ROCT/harm budget after seeing certificate/test results.

For final paper archival after this engineering recovery is stable:
  FORCE_RERUN_VALID_ARMS=1 bash OC-RAP-v48.45.6-stage-isolation-IO-fast-resume-run-commands-ZH.txt
This makes all four arms come from the exact v48.45.6 checkout; it is not needed just to
recover B/C/D and determine whether SOWR has an algorithmic effect.
EOF
