四、准备干净输出目录并安全复用旧 teacher index（推荐）
------------------------------------------------------
OLD_RUN=runs/ocrap_v48_36_ocaf_dedicated_4836
OUTPUTDIR=runs/ocrap_v48_36_1_ocaf_cuda_hotfix_48361
mkdir -p "$OUTPUTDIR"

# 只复制 index 与 summary；不要复制旧 checkpoint、完成状态或 failure marker。
for f in \
  evidence_adapt_teacher_pcd_index.jsonl \
  evidence_adapt_teacher_pcd_index_summary.json \
  evidence_adapt_dev_teacher_pcd_index.jsonl \
  evidence_adapt_dev_teacher_pcd_index_summary.json; do
  if [[ -f "$OLD_RUN/$f" ]]; then cp -f "$OLD_RUN/$f" "$OUTPUTDIR/$f"; fi
done

# 复制后更新 summary 中的 output provenance；不改标签、manifest SHA 或阈值。
python - "$OUTPUTDIR" <<'PY_REBASE_INDEX_SUMMARY'
import json,os,pathlib,sys,time
root=pathlib.Path(sys.argv[1])
for summary_name,index_name in (
    ('evidence_adapt_teacher_pcd_index_summary.json','evidence_adapt_teacher_pcd_index.jsonl'),
    ('evidence_adapt_dev_teacher_pcd_index_summary.json','evidence_adapt_dev_teacher_pcd_index.jsonl'),
):
    path=root/summary_name
    if not path.is_file():
        continue
    doc=json.loads(path.read_text())
    doc['output']=str(root/index_name)
    doc['copied_for_hotfix_rerun']=True
    doc['copied_for_hotfix_rerun_unix']=time.time()
    tmp=path.with_name(f'.{path.name}.tmp.{os.getpid()}')
    tmp.write_text(json.dumps(doc,ensure_ascii=False,indent=2)+'\n')
    os.replace(tmp,path)
PY_REBASE_INDEX_SUMMARY

六、读取权威状态
----------------
python -m json.tool "$OUTPUTDIR/AUTHORITATIVE_RUN_STATUS.json"
python -m json.tool "$OUTPUTDIR/V48_36_COMPLETE.json"

结果解释：
- RC=0：pipeline 有效且 gate 通过，执行 NEXT_COMMANDS.txt。
- RC=20：pipeline 有效但自然 gate 拒绝，可开始算法缺陷分析。
- RC=30：仍是工程失败，优先查看 PIPELINE_FAILED.json 和 FAILURE_SIGNATURE_*.json。

--------------------------------
