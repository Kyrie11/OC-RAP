# OC-RAP v48.39 DRFR — 原实验机执行指令
# 目标：修复 v48.38 zero-update metric contract + reserve train/deploy alignment，
#       并检验 benefit/harm signed physical factor 动态范围是否是剩余 gate 瓶颈。
# D=主实验；A/B/C 在主实验后并发。只有 authoritative RC=0 才执行 test closed-loop。
set -Eeuo pipefail

# ---------------------------------------------------------------------------
# 0. 路径。若你的机器路径不同，只改本节。
# ---------------------------------------------------------------------------
REPO=/home/senzeyu2/code/OC-RAP-v48.39-DRFR
OLD_REPO=/home/senzeyu2/code/OC-RAP
RUN_ROOT="$OLD_REPO/runs"
OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP
PROTOCOL_ROOT="$OCRAP_ROOT/calibration_v48_14_prism_4814"
CAL_SAFE="$OCRAP_ROOT/calibration_safe"
SOURCE_RUN="$OLD_REPO/runs/ocrap_v48_13_terra_proxy_4801"
SAFE_WOMD_SOURCE=/data0/senzeyu2/dataset/WOMD/waymo_open_dataset_motion_v_1_3_1/uncompressed/tf_example/validation/validation_tfexample.tfrecord@150
GPU0=0
GPU1=1

cd "$REPO"
export OCRAP_REPO="$REPO"
export PYTHONPATH="$REPO/src${PYTHONPATH:+:$PYTHONPATH}"
export OCRAP_ROOT PROTOCOL_ROOT CAL_SAFE SOURCE_RUN SAFE_WOMD_SOURCE GPU0 GPU1

# 禁止把 v48.38 的 bounded factor cache 当成 v48.39 结果。
unset V4836_FACTOR_CACHE_BALANCED V4836_FACTOR_CACHE_PRECISION || true
unset V4839_ALLOW_EXACT_FACTOR_CACHE || true

# ---------------------------------------------------------------------------
# 1. 静态/定向预检。不读 test split。
# ---------------------------------------------------------------------------
python -m compileall -q src tools
bash -n scripts/run_v48_39_drfr_dedicated.sh
bash -n scripts/run_v48_39_drfr_ablation_arm.sh
bash -n scripts/run_v48_39_drfr_ablations_parallel.sh
python -m pytest -q \
  tests/test_v48_39_drfr.py \
  tests/test_v48_38_rfr.py \
  tests/test_v48_37_haf.py

# ---------------------------------------------------------------------------
# 2. 主实验 D：unbounded benefit + unbounded harm + aligned reserve。
# ---------------------------------------------------------------------------
MAIN_OUT="$RUN_ROOT/ocrap_v48_39_drfr_main"
if [[ -e "$MAIN_OUT/AUTHORITATIVE_RUN_STATUS.json" || -e "$MAIN_OUT/ATTEMPT_STARTED.json" ]]; then
  echo "REFUSE: $MAIN_OUT 已存在。不要覆盖历史终态；请把 MAIN_OUT 改成新的目录名。" >&2
  exit 2
fi

set +e
OUTPUTDIR="$MAIN_OUT" GPU0="$GPU0" GPU1="$GPU1" \
  bash scripts/run_v48_39_drfr_dedicated.sh
MAIN_SHELL_RC=$?
set -e

echo "v48.39 DRFR main shell RC=$MAIN_SHELL_RC"
[[ -f "$MAIN_OUT/AUTHORITATIVE_RUN_STATUS.json" ]] || {
  echo "ERROR: 缺失 AUTHORITATIVE_RUN_STATUS.json，属于工程失败。" >&2
  exit 30
}
python -m json.tool "$MAIN_OUT/AUTHORITATIVE_RUN_STATUS.json"
[[ -f "$MAIN_OUT/GATE_FAILURE_DECOMPOSITION.json" ]] && \
  python -m json.tool "$MAIN_OUT/GATE_FAILURE_DECOMPOSITION.json"

# 终端断连后也以 authoritative 文件为准；shell RC 只做一致性检查。
MAIN_AUTH_RC="$(python - "$MAIN_OUT/AUTHORITATIVE_RUN_STATUS.json" <<'PY'
import json,sys
x=json.load(open(sys.argv[1]))
assert x.get('valid') is True, x
print(int(x['authoritative_exit_code']))
PY
)"
echo "v48.39 authoritative RC=$MAIN_AUTH_RC"

# 只允许 0/20 进入算法分析与消融。其余先处理 pipeline。
if [[ "$MAIN_AUTH_RC" != 0 && "$MAIN_AUTH_RC" != 20 ]]; then
  echo "STOP: authoritative RC=$MAIN_AUTH_RC 是工程/协议失败。" >&2
  echo "不要跑 test closed-loop；请上传完整 $MAIN_OUT。" >&2
  exit "$MAIN_AUTH_RC"
fi

python - "$MAIN_OUT/AUTHORITATIVE_RUN_STATUS.json" <<'PY'
import json,sys
x=json.load(open(sys.argv[1]))
assert x.get('pipeline_valid') is True, x
assert int(x.get('authoritative_exit_code')) in (0,20), x
print('pipeline-valid scientific terminal:', x['authoritative_exit_code'])
PY

# ---------------------------------------------------------------------------
# 3. A/B/C 动态范围消融：并发运行，D 已由 MAIN_OUT 覆盖。
#    A = bounded benefit + bounded harm
#    B = bounded benefit + unbounded harm
#    C = unbounded benefit + bounded harm
#    D = unbounded benefit + unbounded harm (MAIN)
# ---------------------------------------------------------------------------
for ARM in A B C; do
  OUT="$RUN_ROOT/ocrap_v48_39_drfr_ablation_${ARM}"
  if [[ -e "$OUT/AUTHORITATIVE_RUN_STATUS.json" || -e "$OUT/ATTEMPT_STARTED.json" ]]; then
    echo "REFUSE: $OUT 已存在。请使用新的 RUN_ROOT/目录名，不自动删除旧实验。" >&2
    exit 2
  fi
done

# 三个 arm 同时启动。每个 controller 内部 Balanced→GPU0、Precision→GPU1。
# 默认限制 host thread/loader，避免 3 个 arm/GPU 时 CPU/I/O 过订阅。
BASE_OUT="$RUN_ROOT" GPU0="$GPU0" GPU1="$GPU1" \
ABLATION_OMP_NUM_THREADS=1 \
ABLATION_NUM_WORKERS=2 \
ABLATION_PREFETCH_FACTOR=2 \
  bash scripts/run_v48_39_drfr_ablations_parallel.sh

python -m json.tool "$RUN_ROOT/v48_39_parallel_logs/ABLATION_PARALLEL_SUMMARY.json"
for ARM in A B C; do
  OUT="$RUN_ROOT/ocrap_v48_39_drfr_ablation_${ARM}"
  echo "================ ABLATION $ARM ================"
  python -m json.tool "$OUT/AUTHORITATIVE_RUN_STATUS.json"
  [[ -f "$OUT/GATE_FAILURE_DECOMPOSITION.json" ]] && \
    python -m json.tool "$OUT/GATE_FAILURE_DECOMPOSITION.json"
done

# 若需要独立重复 D 才设置 RERUN_D=1；正常情况下不要重复主实验：
# BASE_OUT="$RUN_ROOT" GPU0=0 GPU1=1 RERUN_D=1 \
#   bash scripts/run_v48_39_drfr_ablations_parallel.sh

# ---------------------------------------------------------------------------
# 4. Closed-loop：只允许主实验 authoritative RC=0。
# ---------------------------------------------------------------------------
if [[ "$MAIN_AUTH_RC" == 0 ]]; then
  [[ -s "$MAIN_OUT/NEXT_COMMANDS.txt" ]] || {
    echo "ERROR: RC=0 但 NEXT_COMMANDS.txt 缺失，属于终态契约错误；禁止手工绕过。" >&2
    exit 30
  }
  echo "Natural gate passed. Authorized closed-loop commands:"
  cat "$MAIN_OUT/NEXT_COMMANDS.txt"
  bash "$MAIN_OUT/NEXT_COMMANDS.txt"
else
  echo "Main RC=20：pipeline 有效但 Natural gate 未通过。"
  echo "不执行 test closed-loop，不手工绕过 NEXT_COMMANDS 授权。"
fi
