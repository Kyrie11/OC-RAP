# 是否分别改善 harm/benefit 的结构可识别性。三个 regime 仍仅作为评估切片，算法无 regime routing。

cd /home/senzeyu2/code/OC-RAP-v48.43-POET
export OCRAP_REPO="$PWD"
export PYTHONPATH="$PWD/src${PYTHONPATH:+:$PYTHONPATH}"

# 修改成你的正式 runs 根目录。
BASE_OUT=/home/senzeyu2/code/OC-RAP/runs
mkdir -p "$BASE_OUT"

# 建议先做轻量静态/CPU preflight（不会读 test root）。
python -m compileall -q src tools tests
bash -n scripts/run_v48_43_poet_ablation_arm.sh
bash -n scripts/run_v48_43_poet_2x2_parallel.sh
PYTHONPATH="$PWD/src" pytest -q \
  tests/test_v48_40_dcfr.py \
  tests/test_v48_41_fcfr.py \
  tests/test_v48_42_hpfr.py \
  tests/test_v48_42_1_metric_provenance_hotfix.py \
  tests/test_v48_43_poet.py

# ------------------------------------------------------------
# 主推荐：A/B/C/D 四个 arm 同时启动。
# 每个 arm 内：Balanced -> GPU0，Precision -> GPU1。
# 因此 MAX_PARALLEL_ARMS=4 时，每张 GPU 最多同时有 4 个训练进程。
# A30/显存足够时优先使用；若 OOM，改为 MAX_PARALLEL_ARMS=2，
# 会自动分两波运行，但算法和统计语义完全不变。
# ------------------------------------------------------------
GPU0=0 GPU1=1 \
BASE_OUT="$BASE_OUT" \
MAX_PARALLEL_ARMS=4 \
ABLATION_OMP_NUM_THREADS=1 \
ABLATION_NUM_WORKERS=1 \
ABLATION_PREFETCH_FACTOR=2 \
bash scripts/run_v48_43_poet_2x2_parallel.sh
LAUNCHER_RC=$?
echo "launcher_rc=$LAUNCHER_RC"

# 如果上面发生 CUDA OOM，仅用下面命令原样重跑未完成 arm；不要改 batch/loss/gate：
# GPU0=0 GPU1=1 BASE_OUT="$BASE_OUT" MAX_PARALLEL_ARMS=2 \
# ABLATION_OMP_NUM_THREADS=1 ABLATION_NUM_WORKERS=1 ABLATION_PREFETCH_FACTOR=2 \

A_OUT="$BASE_OUT/ocrap_v48_43_poet_ablation_A"
B_OUT="$BASE_OUT/ocrap_v48_43_poet_ablation_B"
C_OUT="$BASE_OUT/ocrap_v48_43_poet_ablation_C"
D_OUT="$BASE_OUT/ocrap_v48_43_poet_main"

# ------------------------------------------------------------
# 只认 AUTHORITATIVE_RUN_STATUS.json，不用 launcher 最后的 shell $? 归因算法。
# ------------------------------------------------------------
for RUN in "$A_OUT" "$B_OUT" "$C_OUT" "$D_OUT"; do
  echo "===== $RUN ====="
  python -m json.tool "$RUN/AUTHORITATIVE_RUN_STATUS.json"
done

# RC 解释：
# - RC=30: 工程/协议失败；该 arm 不做算法归因，先修 named failure stage。
# - RC=20: 有效 Natural-gate 算法结果；不要读 held-out test，继续完成 2×2 归因。
# - RC=0 : 仍要完成 A/B/C/D 作为论文消融；仅 D/Main 可按独立 controller 的 NEXT_COMMANDS.txt 进入后续评估。

# ------------------------------------------------------------
# 在四个 arm 均为有效 RC=0/20 后，生成不读取 test root 的 2×2 结构化归因。
# ------------------------------------------------------------
python tools/compare_v48_43_poet_2x2.py \
  --a "$A_OUT" \
  --b "$B_OUT" \
  --c "$C_OUT" \
  --d "$D_OUT" \
  --output "$BASE_OUT/ocrap_v48_43_poet_2x2_comparison.json" \
  > "$BASE_OUT/ocrap_v48_43_poet_2x2_comparison.stdout.json"

# 可选：每个 arm 再输出 component-level frontier 诊断（Precision）。
for RUN in "$A_OUT" "$B_OUT" "$C_OUT" "$D_OUT"; do
  python tools/summarize_v48_43_poet_frontier_diagnostics.py \
    --outputdir "$RUN" \
    --variant precision \
    --output "$RUN/V48_43_PRECISION_FRONTIER_DIAGNOSTICS.json" \
    > /dev/null
done

# ------------------------------------------------------------
# 预注册的算法判定逻辑（不要实验后换指标）：
# ------------------------------------------------------------
# B vs A（harm-side POET）
# 1) Near deployability safe-positive false-veto 必须有“实质”下降，不能只是 16/18 -> 15/18 这类微移；
# 2) harmful-vs-safe-positive deployability AUC 必须同步上升；
# 3) harmful false-safe / harmful-selected UCB 不能用明显恶化来换 recall。
# 建议作为强 go 信号：false-veto <= 12/18 且 deployability conditional AUC >= 0.56；
# 这只是本轮算法 go/no-go 标尺，不是论文最终 gate。
#
# C vs A（benefit-side POET）
# 1) Contact candidate/proposal safe-positive 或 positive AUC 应有 >=~0.05 的结构性提升；
# 2) certificate positive recall 至少进入 0.20 附近（20 个 proposal safe-positive groups 中 >=4 个）；
# 3) harmful-selected UCB 不得恶化。
#
# D/Main
# 1) 同时保留 B 的 Near 低 harmful contamination 和 C 的 Contact benefit gain；
# 2) Contact recall >=0.20 且 harmful-selected UCB 至少不高于 A，最好明显下降；
# 3) shared-development nearest rule 在 Near 和 Contact 都必须开始选择 safe-positive，
#    不能再出现“constraint deficit 下降但两个 stratum positive_selected 都是 0”的假改善；
# 4) 最终目标仍是 shared development rule valid=true / authoritative RC=0。
#
# 若 B/C/D 均不满足以上结构性判据：
# - 不增加 POET 宽度，不调大 transport scale，不再堆 rank/harm losses；
# - 下一步转向 recovery-option conflict / observation-teacher calibration 的可识别性诊断。

# ------------------------------------------------------------
# 只有 D/Main authoritative RC=0 时：
# ------------------------------------------------------------
if [[ -f "$D_OUT/AUTHORITATIVE_RUN_STATUS.json" ]] && \
   python - "$D_OUT/AUTHORITATIVE_RUN_STATUS.json" <<'PY'
import json,sys
x=json.load(open(sys.argv[1]))
raise SystemExit(0 if int(x.get('authoritative_exit_code',-1)) == 0 else 1)
PY
then
  echo "D/Main RC=0. Run ONLY controller-authorized next commands:"
  cat "$D_OUT/NEXT_COMMANDS.txt"
  # 人工确认路径后执行：
  # bash "$D_OUT/NEXT_COMMANDS.txt"
fi
