OC-RAP v50.1 完整外部 Baseline、OC-RAP 三 Regime、对比表和 10 条 MP4 指令
================================================================================

0. 解压并进入仓库
-----------------
unzip OC-RAP-v50.1-full-regime-runtime-fixed.zip
cd OC-RAP-v50.1
export PYTHONPATH="$PWD/src${PYTHONPATH:+:$PYTHONPATH}"

1. 设置数据路径（@150 必须保留）
--------------------------------
export OCRAP_ROOT="/data0/senzeyu2/dataset/OCRAP"
export WOMD_TF_ROOT="/data0/senzeyu2/dataset/WOMD/waymo_open_dataset_motion_v_1_3_1/uncompressed/tf_example"
export WOMD_VAL="$WOMD_TF_ROOT/validation/validation_tfexample.tfrecord@150"
export WOMD_VAL_INTERACTIVE="$WOMD_TF_ROOT/validation_interactive/validation_interactive_tfexample.tfrecord@150"

# 不要使用 /absolute/path/...；不要去掉 @150。

2. 强烈建议先做预检
-------------------
python tools/validate_womd_spec.py --spec "$WOMD_VAL" \
  --output runs/preflight/womd_validation.json
python tools/validate_womd_spec.py --spec "$WOMD_VAL_INTERACTIVE" \
  --output runs/preflight/womd_validation_interactive.json

python tools/check_closed_loop_dataset_support.py \
  --dataset "$OCRAP_ROOT/test_safe" --split test --womd "$WOMD_VAL" \
  --output runs/preflight/test_safe.json
python tools/check_closed_loop_dataset_support.py \
  --dataset "$OCRAP_ROOT/test_near_contact" --split test --womd "$WOMD_VAL_INTERACTIVE" \
  --output runs/preflight/test_near_contact.json
python tools/check_closed_loop_dataset_support.py \
  --dataset "$OCRAP_ROOT/test_contact" --split test --womd "$WOMD_VAL_INTERACTIVE" \
  --output runs/preflight/test_contact.json

# 三个命令都必须返回 0，且 JSON 中：
# valid/schema_supports_closed_loop=true
# num_resolved_womd_files=150

------------------------------------
# 推荐把可复用 checkpoint 放在一个稳定目录，不要绑在临时 OUT 下：
export EXTERNAL_CHECKPOINT_ROOT="/data0/senzeyu2/checkpoints/ocrap_external_baselines_v49"

# 期望布局：
# $EXTERNAL_CHECKPOINT_ROOT/safe/wayformer_bc/best.pt

# 已存在且通过 contract 校验的 checkpoint 会直接复用；缺失的才训练。

13. 常见错误检查
----------------
# A. 检查是否仍有 placeholder：
grep -R "/absolute/path" -n scripts/*.sh *.txt 2>/dev/null || true

# B. 检查 shard 数：
python tools/validate_womd_spec.py --spec "$WOMD_VAL"
python tools/validate_womd_spec.py --spec "$WOMD_VAL_INTERACTIVE"

# C. 检查 external 缺少哪个方法：
python - <<'PY'
import json
p='runs/all_regime_external_baselines_v50_1_full/EXTERNAL_BASELINE_RUN_INDEX.json'
x=json.load(open(p))
print('complete=',x['complete'])
for r,d in x['regimes'].items():
    print(r,d['phase'])
    for m in d['closed_loop_methods']:
        if not m['complete']:
            print('  INCOMPLETE',m)
PY

# D. 检查 OC-RAP 缺少哪个 regime：
python - <<'PY'
import json
p='runs/ocrap_three_regime_closed_loop_v50_1_full/OCRAP_THREE_REGIME_RUN_INDEX.json'

print('complete=',x['complete'])
for r,d in x['regimes'].items(): print(r,d)
PY
