# v48.16 ANCHOR：严格按顺序执行

cd /path/to/OC-RAP-v48.16-ANCHOR
export PYTHONPATH="$PWD/src${PYTHONPATH:+:$PYTHONPATH}"
OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP
PROTOCOL_ROOT=$OCRAP_ROOT/calibration_v48_14_prism_4814

# 0. 数据与WOMD工程预检
python tools/audit_dedicated_protocol_v48_16.py \
  --protocol-root "$PROTOCOL_ROOT" \
  --output runs/v48_16_dedicated_protocol_audit.json

python tools/validate_womd_shards_v48_16.py \
  --source /data0/senzeyu2/dataset/WOMD/waymo_open_dataset_motion_v_1_3_1/uncompressed/tf_example/validation/validation_tfexample.tfrecord@150 \
  --expected-shards 150

# 1. 先修复并重跑已训练v48.15的certificate，不重新训练
set +e
OUTPUTDIR=runs/ocrap_v48_15_prism_cc_dedicated_4815 \
OCRAP_ROOT="$OCRAP_ROOT" \
PROTOCOL_ROOT="$PROTOCOL_ROOT" \
CAL_SAFE="$OCRAP_ROOT/calibration_safe" \
GPU0=0 GPU1=1 \
  bash scripts/recover_v48_15_certificate_pool_v48_16.sh
V4815_CERT_RC=$?
set -e
echo "v48.15 repaired certificate rc=$V4815_CERT_RC"

python tools/check_v48_16_learning_gates.py \
  --run runs/ocrap_v48_15_prism_cc_dedicated_4815 \
  --output runs/ocrap_v48_15_prism_cc_dedicated_4815/learning_gates_repaired_v48_16.json

# rc=0：可以运行授权后的stress closed-loop
if [[ "$V4815_CERT_RC" == 0 ]]; then
  OUT=runs/ocrap_v48_15_prism_cc_dedicated_4815 \
  GPU0=0 GPU1=1 CL_RESUME=0 \
    bash scripts/run_v48_16_stress_if_authorized.sh
fi

# rc=20：是真实gate拒绝，再运行v48.16主实验
# rc=30：先修工程错误，不要训练或闭环。

# 2. Safe paired重新运行：@150且完整扫描validation，至少120个匹配target
BASE_RUN=runs/ocrap_v48_13_terra_proxy_4801/candidates/precision \
RUN=runs/ocrap_v48_16_safe_paired_calibration \
SAFE_TEST="$OCRAP_ROOT/calibration_safe" \
SAFE_WOMD_SOURCE=/data0/senzeyu2/dataset/WOMD/waymo_open_dataset_motion_v_1_3_1/uncompressed/tf_example/validation/validation_tfexample.tfrecord@150 \
SAFE_RAW_MAX_SCENARIOS=0 \
RUN_SAFE_PAIRED_SCALAR=1 \
GPU_SAFE_BASELINE=0 GPU_SAFE=1 \
SAFE_MAX_TARGETS=120 SAFE_MAX_ROLLOUTS=120 \
CL_RESUME=0 \
  bash scripts/run_v48_7_safe_noninferiority.sh

# 3. v48.16主实验（使用dedicated adaptation/dev/certificate，不再使用val做代理calibration）
set +e
OUTPUTDIR=runs/ocrap_v48_16_anchor_dedicated_4816 \
SOURCE_RUN=runs/ocrap_v48_13_terra_proxy_4801 \
OCRAP_ROOT="$OCRAP_ROOT" \
PROTOCOL_ROOT="$PROTOCOL_ROOT" \
CAL_SAFE="$OCRAP_ROOT/calibration_safe" \
GPU0=0 GPU1=1 \
BATCH_SIZE=72 NUM_WORKERS=6 PREFETCH_FACTOR=2 \
EVIDENCE_ADAPT_EPOCHS=12 EVIDENCE_ADAPT_PATIENCE=4 \
EVIDENCE_ADAPT_LR=0.00015 \
EVIDENCE_CALIBRATOR_HIDDEN=4 \
EVIDENCE_CALIBRATOR_SCALE=0.20 \
ORDINAL_EVIDENCE_CLASS_BALANCED_WEIGHT=1.00 \
ORDINAL_EVIDENCE_BENEFIT_MARGIN_WEIGHT=0.75 \
ORDINAL_EVIDENCE_HARM_MARGIN_WEIGHT=1.00 \
ORDINAL_EVIDENCE_TARGET_PROBABILITY=0.60 \
EVIDENCE_CALIBRATOR_ANCHOR_WEIGHT=0.10 \
ORDINAL_EVIDENCE_HARD_HARM_WEIGHT=0.50 \
ORDINAL_EVIDENCE_HARD_BENEFIT_WEIGHT=0.75 \
  bash scripts/run_v48_16_anchor_dedicated.sh \
  > runs/ocrap_v48_16_anchor_dedicated_4816.controller.log 2>&1
V4816_RC=$?
set -e
echo "v48.16 rc=$V4816_RC"

python tools/check_v48_16_learning_gates.py \
  --run runs/ocrap_v48_16_anchor_dedicated_4816 \
  --output runs/ocrap_v48_16_anchor_dedicated_4816/learning_gates_v48_16.json

# 4. 只有rc=0和NEXT_COMMANDS存在时运行stress closed-loop
if [[ "$V4816_RC" == 0 && -s runs/ocrap_v48_16_anchor_dedicated_4816/NEXT_COMMANDS.txt ]]; then
  OUT=runs/ocrap_v48_16_anchor_dedicated_4816 \
  GPU0=0 GPU1=1 CL_RESUME=0 \
    bash scripts/run_v48_16_stress_if_authorized.sh
fi

# 5. 四组消融；每个variant内四组同时运行，每张A30两个轻量任务
nohup env \
ABLATION_ROOT=runs/ocrap_v48_16_ablations \
SOURCE_RUN=runs/ocrap_v48_13_terra_proxy_4801 \
OCRAP_ROOT="$OCRAP_ROOT" \
PROTOCOL_ROOT="$PROTOCOL_ROOT" \
CAL_SAFE="$OCRAP_ROOT/calibration_safe" \
GPU0=0 GPU1=1 \
BATCH_SIZE=72 NUM_WORKERS=3 PREFETCH_FACTOR=2 \
EVIDENCE_ADAPT_EPOCHS=12 EVIDENCE_ADAPT_PATIENCE=4 \
bash scripts/run_v48_16_parallel_ablations.sh \
> runs/ocrap_v48_16_ablations.controller.log 2>&1 </dev/null &
