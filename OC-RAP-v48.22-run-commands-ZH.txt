# OC-RAP v48.22 COVENANT-BRIDGE 执行指令
# 约定：以下命令在项目根目录执行；不要复用 v48.21 输出目录。

cd /path/to/OC-RAP-v48.22-COVENANT-BRIDGE
export PYTHONPATH="$PWD/src${PYTHONPATH:+:$PYTHONPATH}"
export PYTHONUNBUFFERED=1

# 根据服务器实际路径确认。默认值与现有实验保持一致。
export OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP
export SOURCE_RUN=runs/ocrap_v48_13_terra_proxy_4801
export PROTOCOL_ROOT=/data0/senzeyu2/dataset/OCRAP/calibration_v48_14_prism_4814
export CAL_SAFE=/data0/senzeyu2/dataset/OCRAP/calibration_safe
export GPU0=0
export GPU1=1

# ============================================================
# 1. 主实验：Balanced 在 GPU0，Precision 在 GPU1
# ============================================================
OUT=runs/ocrap_v48_22_covenant_dedicated_4822
mkdir -p "$OUT/logs"
set +e
OUTPUTDIR="$OUT" \
OCRAP_ROOT="$OCRAP_ROOT" SOURCE_RUN="$SOURCE_RUN" \
PROTOCOL_ROOT="$PROTOCOL_ROOT" CAL_SAFE="$CAL_SAFE" \
GPU0="$GPU0" GPU1="$GPU1" \
NUM_WORKERS=3 PREFETCH_FACTOR=2 BATCH_SIZE=72 \
EVIDENCE_ADAPT_EPOCHS=24 EVIDENCE_ADAPT_PATIENCE=7 \
bash scripts/run_v48_22_covenant_dedicated.sh \
  2>&1 | tee "$OUT/controller.log"
RC=${PIPESTATUS[0]}
set -e
printf 'v48.22 main rc=%s\n' "$RC"

# 返回码解释：
# RC=0  : independent Natural gate passed；NEXT_COMMANDS.txt 应由 controller 自动生成。
# RC=20 : pipeline/certificate 有效，但算法未通过；不读取 test，执行第 3 节消融。
# RC=30 : 工程/协议/数据/index/checkpoint/certificate 失败；先执行第 2 节排查。

# ============================================================
# 2. RC=30 工程排查（不能做算法结论）
# ============================================================
if [[ "$RC" == 30 ]]; then
  cat "$OUT/PIPELINE_FAILED.json" 2>/dev/null || true
  cat "$OUT/V48_22_COMPLETE.json" 2>/dev/null || true
  tail -n 200 "$OUT/controller.log" 2>/dev/null || true
  find "$OUT/logs" -maxdepth 1 -type f -print -exec sh -c 'echo "--- $1"; tail -n 100 "$1"' _ {} \;
fi

# 若 target/index contract 变化，controller 会自动重建 teacher index。
# 仅在明确需要强制重建时使用：
# REBUILD_ADAPT_INDEX=1 ... bash scripts/run_v48_22_covenant_dedicated.sh

# ============================================================
# 3. 仅当主实验 RC=20：8 个消融一次全部启动
#    GPU0 四任务，GPU1 四任务；不是两波运行。
# ============================================================
if [[ "$RC" == 20 ]]; then
  ABL=runs/ocrap_v48_22_covenant_ablations
  mkdir -p "$ABL/logs"
  set +e
  ABLATION_ROOT="$ABL" \
  OCRAP_ROOT="$OCRAP_ROOT" SOURCE_RUN="$SOURCE_RUN" \
  PROTOCOL_ROOT="$PROTOCOL_ROOT" CAL_SAFE="$CAL_SAFE" \
  GPU0="$GPU0" GPU1="$GPU1" \
  NUM_WORKERS=1 PREFETCH_FACTOR=2 BATCH_SIZE=48 \
  OMP_NUM_THREADS=2 MKL_NUM_THREADS=2 OPENBLAS_NUM_THREADS=2 \
  EVIDENCE_ADAPT_EPOCHS=24 EVIDENCE_ADAPT_PATIENCE=7 \
  bash scripts/run_v48_22_parallel_ablations.sh \
    2>&1 | tee "$ABL/controller.log"
  ABL_RC=${PIPESTATUS[0]}
  set -e
  printf 'v48.22 ablations rc=%s\n' "$ABL_RC"
  cat "$ABL/TASK_GPU_ASSIGNMENT.txt" 2>/dev/null || true
  cat "$ABL/ABLATIONS_STATUS.json" 2>/dev/null || true
fi

# 脚本内固定的轮询分配顺序：
# GPU0: A_two_head_safe_probability_balanced
#       C_triad_group_mil_aggregate_balanced
#       A_two_head_safe_probability_precision
#       C_triad_group_mil_aggregate_precision
# GPU1: B_triad_candidate_only_balanced
#       D_full_covenant_balanced
#       B_triad_candidate_only_precision
#       D_full_covenant_precision
# 实际分配以 TASK_GPU_ASSIGNMENT.txt 为准。

# ============================================================
# 4. 仅当 RC=0：运行授权 stress/closed-loop
# ============================================================
if [[ "$RC" == 0 ]]; then
  test -f "$OUT/NEXT_COMMANDS.txt"
  cat "$OUT/NEXT_COMMANDS.txt"
  OUT="$OUT" GPU0="$GPU0" GPU1="$GPU1" CL_RESUME=0 \
    bash scripts/run_v48_22_stress_if_authorized.sh \
    2>&1 | tee "$OUT/stress_controller.log"
fi

# ============================================================
# 5. 实验结束后必须保存的归因工件
# ============================================================
# 主实验：
#   V48_22_COMPLETE.json
#   COVENANT_TARGET_SUPPORT.json
#   COVENANT_INDEX_CONTRACT.json
#   learning_gates_v48_22.json
#   candidates/{balanced,precision}/model_v48_trac_sr/train_summary.json
#   candidates/{balanced,precision}/calibration/direct_value_risk_{near,contact}_v48.json
#   candidates/{balanced,precision}/calibration/*.rows.jsonl
#
# 消融：
#   TASK_GPU_ASSIGNMENT.txt
#   ABLATIONS_STATUS.json / ABLATIONS_COMPLETE.json
#   tasks/*/TASK_COMPLETE.json
#   tasks/*/candidates/*/model_v48_trac_sr/train_summary.json
#   tasks/*/candidates/*/calibration/direct_value_risk_{near,contact}_v48.json
#
# 不要在 RC=20 或 RC=30 时运行 stress/test，也不要手工创建 NEXT_COMMANDS.txt。
