#!/usr/bin/env bash
# OC-RAP v48.52 DCP-DRFC-BCDE-PSA — 两 GPU authoritative A/B
# 默认严格复用已验证 v48.51-B 作为 A reference；若复用 contract 不通过则自动 fresh A/B。
# Safe / near_contact / contact 仅为数据/评测 strata，不进入 policy/router/threshold/budget。
set -Eeuo pipefail

REPO=/home/senzeyu2/code/OC-RAP-v48.52-DCP-DRFC-BCDE-PSA
BASE_OUT=/home/senzeyu2/code/OC-RAP/runs
SOURCE_RUN=/home/senzeyu2/code/OC-RAP/runs/ocrap_v48_45_source_rebuild_s7
OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP
V4852_REFERENCE_RUN=/home/senzeyu2/code/OC-RAP/runs/ocrap_v48_51_dcp_drfc_bcde_ablation_B
GPU0=0
GPU1=1

cd "$REPO"
export BASE_OUT SOURCE_RUN OCRAP_ROOT V4852_REFERENCE_RUN GPU0 GPU1

# 默认：若 v48.51-B 的 protocol seal / source SHA / factor contract / authoritative
# 状态全部一致，则只训练 v48.52 PSA Main，Balanced->GPU0、Precision->GPU1 并行。
# 如需显式强制 fresh A/B，可在执行前设置：export V4852_FORCE_FRESH_REFERENCE=1
bash scripts/run_v48_52_dcp_drfc_bcde_psa_two_gpu.sh

AUDIT="$BASE_OUT/OC-RAP-v48.52-DCP-DRFC-BCDE-PSA-AB-audit.json"
MAIN="$BASE_OUT/ocrap_v48_52_dcp_drfc_bcde_psa_main"

echo "===== v48.52 PSA A/B audit ====="
python - "$AUDIT" "$MAIN/AUTHORITATIVE_RUN_STATUS.json" <<'PY'
import json,sys
au=json.load(open(sys.argv[1],encoding='utf-8'))
st=json.load(open(sys.argv[2],encoding='utf-8'))
ac=au.get('attribution_contract') or {}
print('attribution_valid =', ac.get('valid'))
print('reference_source =', ac.get('reference_source'))
print('Main_authoritative_RC =', st.get('authoritative_exit_code'))
print('Main_pipeline_valid =', st.get('pipeline_valid'))
for arm in ('A','B'):
    a=(au.get('arms') or {}).get(arm) or {}
    near=((a.get('near') or {}).get('certificate') or {})
    contact=((a.get('contact') or {}).get('certificate') or {})
    print(arm,
          'RC=',a.get('authoritative_exit_code'),
          'near_recall=',near.get('positive_recall'),
          'near_UCB90=',near.get('harmful_selected_ucb90'),
          'contact_recall=',contact.get('positive_recall'),
          'contact_UCB90=',contact.get('harmful_selected_ucb90'))
PY

echo "下一轮请上传："
echo "1) $BASE_OUT/ocrap_v48_52_dcp_drfc_bcde_psa_main.zip"
echo "2) $BASE_OUT/OC-RAP-v48.52-DCP-DRFC-BCDE-PSA-AB-audit.upload.json"
echo "3) $BASE_OUT/OC-RAP-v48.52-runtime-telemetry-summary.json"
echo "4) $BASE_OUT/OC-RAP-v48.52-runtime-telemetry.jsonl"
echo "若 audit 中 reference_source=fresh-v48.52-A，再额外上传 fresh A ZIP。"
