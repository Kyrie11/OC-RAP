# v48.14 OC-TRAC-PRISM：完整执行顺序

# 0. 进入目录并确认旧任务退出
cd /path/to/OC-RAP-v48.14-PRISM
export PYTHONPATH="$PWD/src${PYTHONPATH:+:$PYTHONPATH}"
ps -ef | grep -E 'ocrap|run_v48|train_ocrap' | grep -v grep || true

# 1. 建立不可变、scene-disjoint dedicated calibration protocol
# 输出：
#   evidence_adapt_train_near_contact / contact
#   evidence_adapt_dev_near_contact / contact
#   certificate_pool_near_contact / contact
python tools/partition_dedicated_calibration_v48_14.py \
  --near /data0/senzeyu2/dataset/OCRAP/calibration_near_contact \
  --contact /data0/senzeyu2/dataset/OCRAP/calibration_contact \
  --output-root /data0/senzeyu2/dataset/OCRAP/calibration_v48_14_prism_4814 \
  --adapt-train-fraction 0.45 \
  --adapt-dev-fraction 0.15 \
  --seed 4814 \
  --link-mode hardlink \
  --overwrite
cat /data0/senzeyu2/dataset/OCRAP/calibration_v48_14_prism_4814/CALIBRATION_PROTOCOL_COMPLETE.json

# 2. 先对原 v48.13 固定 checkpoint 做 dedicated-only baseline。
# 该命令会生成此前缺失的 gamma_rec_by_bucket_v48.json 和
# direct_value_risk_{near,contact}_v48.json，但不会污染原 proxy 运行目录。
set +e
SOURCE_RUN=runs/ocrap_v48_13_terra_proxy_4801 \

  bash scripts/recalibrate_source_v48_14_dedicated.sh \
  > runs/ocrap_v48_13_terra_dedicated_baseline_4814.controller.log 2>&1
BASELINE_RC=$?
set -e
echo "dedicated baseline rc=$BASELINE_RC (20 表示 Natural gate 未通过但诊断完整)"
cat runs/ocrap_v48_13_terra_dedicated_baseline_4814/dedicated_recalibration_status.json

# 3. v48.14 主实验：冻结 v48.13 top-k proposal，仅在 dedicated adaptation scenes
# 上微调轻量 evidence adapters，再在完全独立的 certificate pool 做 fit/verify。

OUTPUTDIR=runs/ocrap_v48_14_prism_dedicated_4814 \

ORDINAL_EVIDENCE_HARD_BENEFIT_WEIGHT=0.75 \
  bash scripts/run_v48_14_prism_dedicated.sh \
  > runs/ocrap_v48_14_prism_dedicated_4814.controller.log 2>&1
MAIN_RC=$?
set -e
echo "v48.14 rc=$MAIN_RC (20 表示 gate 未通过但所有诊断完整)"
cat runs/ocrap_v48_14_prism_dedicated_4814/V48_14_COMPLETE.json
cat runs/ocrap_v48_14_prism_dedicated_4814/dedicated_recalibration_status.json

# 4. 检查关键产物。两种 variant 都应有这些文件。
for v in balanced precision; do
  base=runs/ocrap_v48_14_prism_dedicated_4814/candidates/$v
  test -f "$base/model_v48_trac_sr/best.pt"
  test -f "$base/calibration/gamma_rec_by_bucket_v48.json"
  test -f "$base/calibration/direct_value_risk_near_v48.json"
  test -f "$base/calibration/direct_value_risk_contact_v48.json"
  test -f "$base/calibration/CERTIFICATE_CALIBRATION_COMPLETE.json"
done
